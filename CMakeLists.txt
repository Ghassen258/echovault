cmake_minimum_required(VERSION 3.16)
project(EchoVault VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(echovault_core src/game.cpp)
target_include_directories(echovault_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(echovault_core PRIVATE /W4)
else()
  target_compile_options(echovault_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(echovault src/main.cpp)
target_link_libraries(echovault PRIVATE echovault_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(echovault_tests tests/test_game.cpp)
  target_link_libraries(echovault_tests PRIVATE echovault_core)
  add_test(NAME game_mechanics COMMAND echovault_tests)
  add_test(NAME scripted_demo COMMAND echovault --no-color --script ddssssrssddddddddddddddn)
  add_test(NAME animated_demo COMMAND echovault --no-color --no-animation --demo)
endif()
